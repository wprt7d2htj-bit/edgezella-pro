Supabase SQL Tables & Security Setup

Supabase Dashboard → SQL Editor → New Query mein paste karke Run karein

create table if not exists public.trader_profiles (
  user_id uuid primary key references auth.users(id) on delete cascade,
  starting_balance numeric not null default 10000,
  updated_at timestamptz not null default now()
);

create table if not exists public.trades (
  id text primary key,
  user_id uuid not null references auth.users(id) on delete cascade,
  trade_date text not null,
  trade_time text default '14:30',
  symbol text not null,
  side text not null,
  entry_price text default '',
  exit_price text default '',
  r_multiple text default '',
  pnl numeric not null default 0,
  setup text default '',
  notes text default '',
  created_at timestamptz not null default now()
);

create table if not exists public.playbook_strategies (
  id text primary key,
  user_id uuid not null references auth.users(id) on delete cascade,
  name text not null,
  session text default 'London / NY Overlap',
  target_r text default '1:3',
  rules text default '',
  created_at timestamptz not null default now()
);

create table if not exists public.daily_journals (
  id text primary key,
  user_id uuid not null references auth.users(id) on delete cascade,
  journal_date text not null,
  mood text default 'Disciplined & Calm',
  rating text default '5',
  pre_market text default '',
  post_market text default '',
  updated_at timestamptz not null default now()
);

alter table public.trader_profiles enable row level security;
alter table public.trades enable row level security;
alter table public.playbook_strategies enable row level security;
alter table public.daily_journals enable row level security;

create policy "profiles_owner" on public.trader_profiles for all using (auth.uid() = user_id) with check (auth.uid() = user_id);
create policy "trades_owner" on public.trades for all using (auth.uid() = user_id) with check (auth.uid() = user_id);
create policy "playbook_owner" on public.playbook_strategies for all using (auth.uid() = user_id) with check (auth.uid() = user_id);
create policy "journals_owner" on public.daily_journals for all using (auth.uid() = user_id) with check (auth.uid() = user_id);
